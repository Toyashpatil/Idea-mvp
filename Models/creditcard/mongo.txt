#include <stdio.h>
#include <time.h>
void insertion(int *a, int n)
{
 for(int i = 0; i < n; i++)
 {
 for(int j = i; j > 0; j--)
 {
 if(a[j] < a[j - 1])
 {
 int t = a[j];
 a[j] = a[j - 1];
 a[j - 1] = t;
 }
 else
 break;
 }
 }
}
int main()
{
 int n;
 scanf("%d", &n);
 int a[n];
 for(int i = 0; i < n; i++)
 scanf("%d", &a[i]);
 clock_t start = clock();
 insertion(a, n);
 clock_t end = clock();
 for(int i = 0; i < n; i++)
 printf("%d ", a[i]);
 printf("\n")
 
 double time_ms = (double)(end - start) * 1000.0 / CLOCKS_PER_SEC;
 printf("%.3f\n", time_ms);
 return 0;
}
